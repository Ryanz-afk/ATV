/*
 * Insertion Sort em C
 * Implementação in-place, com demonstração passo a passo, teste de
 * estabilidade e benchmark em 5 cenários.
 *
 * Compilar:  gcc -O2 -Wall -Wextra -o insertion_sort insertion_sort.c
 * Demo:      ./insertion_sort demo
 * Rápido:    ./insertion_sort quick     (tabela pequena para a demo ao vivo)
 * Benchmark: ./insertion_sort bench > resultados_insertion.csv
 * Sem args:  demo + quick
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

/* ------------------------------------------------------------------ */
/* Insertion Sort                                                      */
/* ------------------------------------------------------------------ */

static void print_vet(const int *a, int n, int marca)
{
    /* imprime o vetor; a barra '|' separa a parte ordenada da restante */
    for (int i = 0; i < n; i++) {
        if (i == marca) printf("| ");
        printf("%d ", a[i]);
    }
    printf("\n");
}

/* Ordena a[0..n-1] no próprio vetor. Retorna 0 em sucesso, -1 se a == NULL.
 * Se trace != 0, imprime o vetor após cada inserção. */
int insertion_sort(int *a, int n, int trace)
{
    if (a == NULL || n < 0) return -1;

    for (int i = 1; i < n; i++) {
        int chave = a[i];       /* elemento a ser inserido */
        int j = i - 1;

        /* '>' estrito garante ESTABILIDADE: em empate, a chave não passa
         * à frente de um elemento igual que já estava na parte ordenada */
        while (j >= 0 && a[j] > chave) {
            a[j + 1] = a[j];    /* desloca o maior uma posição à direita */
            j--;
        }
        a[j + 1] = chave;       /* encaixa a chave na posição correta */

        if (trace) {
            printf("i=%d (chave %d): ", i, chave);
            print_vet(a, n, i + 1);
        }
    }
    return 0;
}

/* ------------------------------------------------------------------ */
/* Geradores dos 5 cenários                                            */
/* ------------------------------------------------------------------ */

typedef enum { ALEATORIO, ORDENADO, DECRESCENTE, QUASE_ORDENADO, DUPLICADAS,
               N_CENARIOS } Cenario;

static const char *NOMES[N_CENARIOS] = {
    "aleatorio", "ordenado", "decrescente", "quase_ordenado", "duplicadas"
};

static void gerar(int *a, int n, Cenario c)
{
    switch (c) {
    case ALEATORIO:
        for (int i = 0; i < n; i++) a[i] = rand();
        break;
    case ORDENADO:
        for (int i = 0; i < n; i++) a[i] = i;
        break;
    case DECRESCENTE:
        for (int i = 0; i < n; i++) a[i] = n - i;
        break;
    case QUASE_ORDENADO:
        /* ordenado + pequenas perturbações: n/20 trocas entre posições
         * vizinhas (distância de até 10) */
        for (int i = 0; i < n; i++) a[i] = i;
        for (int t = 0; t < n / 20 + 1; t++) {
            int x = rand() % n;
            int y = x + 1 + rand() % 10;
            if (y >= n) y = n - 1;
            int tmp = a[x]; a[x] = a[y]; a[y] = tmp;
        }
        break;
    case DUPLICADAS:
        /* apenas 100 valores distintos -> muitas chaves repetidas */
        for (int i = 0; i < n; i++) a[i] = rand() % 100;
        break;
    default:
        break;
    }
}

/* ------------------------------------------------------------------ */
/* Utilitários de teste e medição                                      */
/* ------------------------------------------------------------------ */

static int esta_ordenado(const int *a, int n)
{
    for (int i = 1; i < n; i++)
        if (a[i - 1] > a[i]) return 0;
    return 1;
}

static double agora(void)
{
    struct timespec ts;
    clock_gettime(CLOCK_MONOTONIC, &ts);
    return (double)ts.tv_sec + (double)ts.tv_nsec / 1e9;
}

/* Tempo médio (segundos) de 'rep' execuções; -1.0 em caso de erro. */
static double medir(int n, Cenario c, int rep)
{
    if (n <= 0 || rep <= 0) return -1.0;

    int *base = malloc((size_t)n * sizeof(int));
    int *work = malloc((size_t)n * sizeof(int));
    if (!base || !work) {
        free(base); free(work);
        return -1.0;
    }

    double total = 0.0;
    /* r = -1 é uma execução de aquecimento (não entra na média) */
    for (int r = -1; r < rep; r++) {
        gerar(base, n, c);
        memcpy(work, base, (size_t)n * sizeof(int));

        double t0 = agora();
        int rc = insertion_sort(work, n, 0);
        double t1 = agora();

        if (rc != 0 || !esta_ordenado(work, n)) {
            fprintf(stderr, "FALHA: n=%d cenario=%s\n", n, NOMES[c]);
            free(base); free(work);
            return -1.0;
        }
        if (r >= 0) total += t1 - t0;
    }
    free(base);
    free(work);
    return total / rep;
}

/* Teste de estabilidade: ordena pares (chave, id) só pela chave. */
typedef struct { int chave; int id; } Par;

static void insertion_par(Par *a, int n)
{
    for (int i = 1; i < n; i++) {
        Par chave = a[i];
        int j = i - 1;
        while (j >= 0 && a[j].chave > chave.chave) {
            a[j + 1] = a[j];
            j--;
        }
        a[j + 1] = chave;
    }
}

static int testar_estabilidade(void)
{
    enum { N = 1000 };
    Par a[N];
    for (int i = 0; i < N; i++) { a[i].chave = rand() % 10; a[i].id = i; }
    insertion_par(a, N);
    for (int i = 1; i < N; i++)
        if (a[i - 1].chave == a[i].chave && a[i - 1].id > a[i].id)
            return 0;
    return 1;
}

/* ------------------------------------------------------------------ */
/* Modos de execução                                                   */
/* ------------------------------------------------------------------ */

static void modo_demo(void)
{
    int v[] = {38, 27, 43, 3, 9, 82, 10, 3};
    int n = (int)(sizeof v / sizeof v[0]);

    printf("=== Demonstração passo a passo ===\n");
    printf("Vetor original: ");
    print_vet(v, n, -1);
    printf("(a barra '|' separa a parte já ordenada do restante)\n\n");

    insertion_sort(v, n, 1);

    printf("\nVetor ordenado: ");
    print_vet(v, n, -1);
    printf("Ordenado corretamente? %s\n", esta_ordenado(v, n) ? "sim" : "NAO");
    printf("Teste de estabilidade (1000 pares): %s\n\n",
           testar_estabilidade() ? "estavel" : "INSTAVEL");
}

static void modo_quick(void)
{
    const int n = 20000;
    printf("=== Teste rápido: n = %d, média de 3 execuções ===\n", n);
    printf("%-16s %12s\n", "cenario", "tempo (ms)");
    for (int c = 0; c < N_CENARIOS; c++) {
        double t = medir(n, (Cenario)c, 3);
        if (t < 0) return;
        printf("%-16s %12.3f\n", NOMES[c], t * 1000.0);
    }
}

static void modo_bench(void)
{
    const int tamanhos[] = {1000, 5000, 10000, 25000, 50000, 100000};
    const int nt = (int)(sizeof tamanhos / sizeof tamanhos[0]);

    printf("cenario,n,tempo_ms\n");
    for (int c = 0; c < N_CENARIOS; c++) {
        for (int i = 0; i < nt; i++) {
            double t = medir(tamanhos[i], (Cenario)c, 3);
            if (t < 0) exit(EXIT_FAILURE);
            printf("%s,%d,%.4f\n", NOMES[c], tamanhos[i], t * 1000.0);
            fflush(stdout);
        }
    }
}

int main(int argc, char **argv)
{
    srand(12345); /* semente fixa: resultados reproduzíveis */

    if (argc < 2) {
        modo_demo();
        modo_quick();
    } else if (strcmp(argv[1], "demo") == 0) {
        modo_demo();
    } else if (strcmp(argv[1], "quick") == 0) {
        modo_quick();
    } else if (strcmp(argv[1], "bench") == 0) {
        modo_bench();
    } else {
        fprintf(stderr, "Uso: %s [demo|quick|bench]\n", argv[0]);
        return EXIT_FAILURE;
    }
    return EXIT_SUCCESS;
}
